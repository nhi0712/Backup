#[cfg(test)]
mod tests {
    use super::{Stack, Node};

    #[test]
    fn test_push_pop() {
        let mut stack = Stack::new();
        assert!(stack.is_empty());

        stack.push(1);
        stack.push(2);
        stack.push(3);

        assert_eq!(stack.size(), 3);
        assert_eq!(stack.pop(), Some(3));
        assert_eq!(stack.pop(), Some(2));
        assert_eq!(stack.pop(), Some(1));
        assert!(stack.is_empty());
    }

    #[test]
    fn test_peek() {
        let mut stack = Stack::new();
        stack.push(1);
        stack.push(2);
        stack.push(3);

        assert_eq!(stack.peek(), Some(&3));
        stack.pop();
        assert_eq!(stack.peek(), Some(&2));
    }

    #[test]
    fn test_size() {
        let mut stack = Stack::new();
        assert_eq!(stack.size(), 0);

        stack.push(1);
        assert_eq!(stack.size(), 1);

        stack.push(2);
        assert_eq!(stack.size(), 2);

        stack.pop();
        assert_eq!(stack.size(), 1);
    }

    #[test]
    fn test_is_empty() {
        let mut stack = Stack::new();
        assert!(stack.is_empty());

        stack.push(1);
        assert!(!stack.is_empty());

        stack.pop();
        assert!(stack.is_empty());
    }

    #[test]
    fn test_is_full() {
        let mut stack = Stack::new();
        assert!(!stack.is_full());

        stack.push(1);
        assert!(stack.is_full());

        stack.pop();
        assert!(!stack.is_full());
    }

    #[test]
    fn test_to_string() {
        let mut stack = Stack::new();
        stack.push(1);
        stack.push(2);
        stack.push(3);

        assert_eq!(stack.to_string(), "3 -> 2 -> 1");
        stack.pop();
        assert_eq!(stack.to_string(), "2 -> 1");
    }

    #[test]
    fn test_push_multiple() {
        let mut stack = Stack::new();
        stack.push_multiple(vec![1, 2, 3]);

        assert_eq!(stack.size(), 3);
        assert_eq!(stack.pop(), Some(3));
        assert_eq!(stack.pop(), Some(2));
        assert_eq!(stack.pop(), Some(1));
    }

    #[test]
    fn test_equals() {
        let mut stack1 = Stack::new();
        stack1.push(1);
        stack1.push(2);
        stack1.push(3);

        let mut stack2 = Stack::new();
        stack2.push(1);
        stack2.push(2);
        stack2.push(3);

        assert!(stack1.equals(&stack2));

        stack2.pop();
        assert!(!stack1.equals(&stack2));
    }
}
