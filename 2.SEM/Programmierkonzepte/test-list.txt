#[cfg(test)]
mod tests {
    use super::LinkedList; // Importiert die LinkedList aus dem übergeordneten Scope

    #[test]
    fn test_create_empty_list() {
        let list: LinkedList<i32> = LinkedList::new();
        assert!(list.is_empty());
        assert_eq!(list.size(), 0);
    }

    #[test]
    fn test_add_elements() {
        let mut list = LinkedList::new();
        list.add(vec![1, 2, 3]);

        assert_eq!(list.size(), 3);
        assert_eq!(list.get(0), Some(&3));
        assert_eq!(list.get(1), Some(&2));
        assert_eq!(list.get(2), Some(&1));
    }

    #[test]
    fn test_insert_elements() {
        let mut list = LinkedList::new();
        list.add(vec![1, 3]);
        list.insert(2, 1);

        assert_eq!(list.size(), 3);
        assert_eq!(list.get(0), Some(&3));
        assert_eq!(list.get(1), Some(&2));
        assert_eq!(list.get(2), Some(&1));
    }

    #[test]
    fn test_remove_at() {
        let mut list = LinkedList::new();
        list.add(vec![1, 2, 3]);

        let removed = list.remove_at(1);
        assert_eq!(removed, Some(2));
        assert_eq!(list.size(), 2);
        assert_eq!(list.get(1), Some(&1));
    }

    #[test]
    fn test_replace() {
        let mut list = LinkedList::new();
        list.add(vec![1, 2, 3]);

        let replaced = list.replace(4, 1);
        assert_eq!(replaced, Some(2));
        assert_eq!(list.get(1), Some(&4));
    }

    #[test]
    fn test_remove_element() {
        let mut list = LinkedList::new();
        list.add(vec![1, 2, 3]);

        let removed = list.remove(2);
        assert_eq!(removed, Some(2));
        assert_eq!(list.size(), 2);
        assert_eq!(list.get(1), Some(&1));
    }

    #[test]
    fn test_to_string() {
        let mut list = LinkedList::new();
        list.add(vec![1, 2, 3]);

        let representation = list.to_string();
        assert_eq!(representation, "3 -> 2 -> 1");
    }
}
